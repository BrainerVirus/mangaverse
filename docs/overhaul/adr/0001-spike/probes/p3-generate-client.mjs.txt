// Generates the typed client X2 calls for: the same names as @app/db, each running in the host.
// Usage: node p3-generate-client.mjs <db-src-dir> <index-module-name> <out.ts>
import { readFileSync, writeFileSync } from "node:fs";
import path from "node:path";

const [srcDir, indexName, outFile] = process.argv.slice(2);

function collect(file) {
  const text = readFileSync(file, "utf8");
  const names = [...text.matchAll(/^export (?:async )?function (\w+)(?:<[^(]*>)?\(\s*db\b/gm)].map(
    (m) => m[1],
  );
  for (const match of text.matchAll(/^export \* from '(\.[^']+)\.js'/gm)) {
    names.push(...collect(path.resolve(path.dirname(file), `${match[1]}.ts`)));
  }
  for (const match of text.matchAll(/^export \{([^}]+)\} from '(\.[^']+)\.js'/gm)) {
    const local = match[1].split(",").map((name) => name.trim());
    const source = readFileSync(path.resolve(path.dirname(file), `${match[2]}.ts`), "utf8");
    for (const name of local) {
      if (
        new RegExp(`^export (?:async )?function ${name}(?:<[^(]*>)?\\(\\s*db\\b`, "m").test(source)
      )
        names.push(name);
    }
  }
  return names;
}

const names = [...new Set(collect(path.join(srcDir, `${indexName}.ts`)))]
  .filter((name) => name !== "withTransaction")
  .sort();
const lines = [
  "// Generated by p3-generate-client.mjs. Same names as the repositories; each call runs in the host.",
  `import type * as host from './${indexName}.js';`,
  `import type { AppDrizzleDb as HostDb } from './adapter.js';`,
  `export * from './${indexName}.js';`,
  "",
  "/** The handle features pass around today as `db`. It is a connection to the host, not SQLite. */",
  "export interface HostHandle {",
  "  call(method: string, args: readonly unknown[]): Promise<unknown>;",
  "}",
  "export type AppDrizzleDb = HostHandle;",
  "",
  "type Bound<F> = F extends (db: HostDb, ...args: infer A) => infer R",
  "  ? (db: HostHandle, ...args: A) => Promise<Awaited<R>>",
  "  : never;",
  "const bind =",
  "  <F,>(method: string) =>",
  "  ((db: HostHandle, ...args: readonly unknown[]) => db.call(method, args)) as Bound<F>;",
  "",
  ...names.map((name) => `export const ${name} = bind<typeof host.${name}>('${name}');`),
  "",
];
writeFileSync(outFile, lines.join("\n"));
console.log(`generated ${names.length} procedures into ${outFile}`);
